# from tvm.script import ir as I
# from tvm.script import tirx as T

@I.ir_module
class Module:
    @T.prim_func
    def main(A: T.Buffer((1024, 2048), "float32"), B: T.Buffer((2048, 64), "float32"), C: T.Buffer((1024, 64), "float32")):
        T.func_attr({"tir.noalias": True, "tirx.noalias": True})
        # with T.sblock("root"):
        C_local = T.sblock_alloc_buffer((1024, 64), scope="local")
        A_shared = T.sblock_alloc_buffer((1024, 2048), scope="shared")
        B_shared = T.sblock_alloc_buffer((2048, 64), scope="shared")
        for i_0_j_0_fused in T.thread_binding(64, thread="blockIdx.x", annotations={"pragma_auto_unroll_max_step": 256, "pragma_unroll_explicit": 1}):
            for i_1_j_1_fused in T.thread_binding(4, thread="vthread.x"):
                for i_2_j_2_fused in T.thread_binding(256, thread="threadIdx.x"):
                    for i_3_init, j_3_init, i_4_init, j_4_init in T.grid(1, 1, 1, 1):
                        with T.sblock("C_init"):
                            vi = T.axis.spatial(1024, i_0_j_0_fused * 16 + i_1_j_1_fused // 2 * 8 + i_2_j_2_fused // 32 + i_3_init + i_4_init)
                            vj = T.axis.spatial(64, i_1_j_1_fused % 2 * 32 + i_2_j_2_fused % 32 + j_3_init + j_4_init)
                            T.reads()
                            T.writes(C_local[vi, vj])
                            T.sblock_attr({"meta_schedule.thread_extent_high_inclusive": 1024, "meta_schedule.thread_extent_low_inclusive": 32, "meta_schedule.tiling_structure": "SSSRRSRS"})
                            C_local[vi, vj] = T.float32(0.0)
                    for k_0 in range(64):
                        for ax0_ax1_fused_0 in range(2):
                            for ax0_ax1_fused_1 in T.thread_binding(256, thread="threadIdx.x"):
                                with T.sblock("A_shared"):
                                    v0 = T.axis.spatial(1024, i_0_j_0_fused * 16 + (ax0_ax1_fused_0 * 256 + ax0_ax1_fused_1) // 32)
                                    v1 = T.axis.spatial(2048, k_0 * 32 + (ax0_ax1_fused_0 * 256 + ax0_ax1_fused_1) % 32)
                                    T.reads(A[v0, v1])
                                    T.writes(A_shared[v0, v1])
                                    A_shared[v0, v1] = A[v0, v1]
                        for ax0_ax1_fused_0 in range(2):
                            for ax0_ax1_fused_1 in T.thread_binding(256, thread="threadIdx.x"):
                                for ax0_ax1_fused_2 in T.vectorized(4):
                                    with T.sblock("B_shared"):
                                        v0 = T.axis.spatial(2048, k_0 * 32 + (ax0_ax1_fused_0 * 1024 + ax0_ax1_fused_1 * 4 + ax0_ax1_fused_2) // 64)
                                        v1 = T.axis.spatial(64, (ax0_ax1_fused_0 * 1024 + ax0_ax1_fused_1 * 4 + ax0_ax1_fused_2) % 64)
                                        T.reads(B[v0, v1])
                                        T.writes(B_shared[v0, v1])
                                        B_shared[v0, v1] = B[v0, v1]
                        for k_1, i_3, j_3, k_2, i_4, j_4 in T.grid(16, 1, 1, 2, 1, 1):
                            with T.sblock("C_update"):
                                vi = T.axis.spatial(1024, i_0_j_0_fused * 16 + i_1_j_1_fused // 2 * 8 + i_2_j_2_fused // 32 + i_3 + i_4)
                                vj = T.axis.spatial(64, i_1_j_1_fused % 2 * 32 + i_2_j_2_fused % 32 + j_3 + j_4)
                                vk = T.axis.reduce(2048, k_0 * 32 + k_1 * 2 + k_2)
                                T.reads(C_local[vi, vj], A_shared[vi, vk], B_shared[vk, vj])
                                T.writes(C_local[vi, vj])
                                T.sblock_attr({"meta_schedule.thread_extent_high_inclusive": 1024, "meta_schedule.thread_extent_low_inclusive": 32, "meta_schedule.tiling_structure": "SSSRRSRS"})
                                C_local[vi, vj] = C_local[vi, vj] + A_shared[vi, vk] * B_shared[vk, vj]
                    for ax0, ax1 in T.grid(1, 1):
                        with T.sblock("C_local"):
                            v0 = T.axis.spatial(1024, i_0_j_0_fused * 16 + i_1_j_1_fused // 2 * 8 + i_2_j_2_fused // 32 + ax0)
                            v1 = T.axis.spatial(64, i_1_j_1_fused % 2 * 32 + i_2_j_2_fused % 32 + ax1)
                            T.reads(C_local[v0, v1])
                            T.writes(C[v0, v1])
                            C[v0, v1] = C_local[v0, v1]
