# from tvm.script import ir as I
# from tvm.script import tirx as T

@I.ir_module
class Module:
    @T.prim_func
    def main(A: T.Buffer((64, 64), "float32"), B: T.Buffer((64, 64), "float32"), C: T.Buffer((64, 64), "float32")):
        T.func_attr({"tir.noalias": True, "tirx.noalias": True})
        # with T.sblock("root"):
        for i_0 in T.parallel(8):
            for j_0, k_0 in T.grid(8, 8):
                for i_1 in T.unroll(8):
                    for j_1 in T.vectorized(8):
                        for k_1 in range(8):
                            with T.sblock("C"):
                                vi = T.axis.spatial(64, i_0 * 8 + i_1)
                                vj = T.axis.spatial(64, j_0 * 8 + j_1)
                                vk = T.axis.reduce(64, k_0 * 8 + k_1)
                                T.reads(A[vi, vk], B[vk, vj])
                                T.writes(C[vi, vj])
                                with T.init():
                                    C[vi, vj] = T.float32(0.0)
                                C[vi, vj] = C[vi, vj] + A[vi, vk] * B[vk, vj]
