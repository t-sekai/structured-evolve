# from tvm.script import ir as I
# from tvm.script import tirx as T

@I.ir_module
class Module:
    @T.prim_func
    def main(A: T.Buffer((64, 64), "float32"), B: T.Buffer((64, 64), "float32"), C: T.Buffer((64, 64), "float32")):
        T.func_attr({"tir.noalias": True, "tirx.noalias": True})
        # with T.sblock("root"):
        A_global = T.sblock_alloc_buffer((64, 64))
        B_global = T.sblock_alloc_buffer((64, 64))
        C_global = T.sblock_alloc_buffer((64, 64))
        for ax0, ax1 in T.grid(64, 64):
            with T.sblock("B_global"):
                v0, v1 = T.axis.remap("SS", [ax0, ax1])
                T.reads(B[v0, v1])
                T.writes(B_global[v0, v1])
                B_global[v0, v1] = B[v0, v1]
        for ax0, ax1 in T.grid(64, 64):
            with T.sblock("A_global"):
                v0, v1 = T.axis.remap("SS", [ax0, ax1])
                T.reads(A[v0, v1])
                T.writes(A_global[v0, v1])
                A_global[v0, v1] = A[v0, v1]
        for i_0 in T.parallel(16):
            for j_0, i_1 in T.grid(16, 4):
                for j_1 in T.vectorized(4):
                    for k in range(64):
                        with T.sblock("C"):
                            vi = T.axis.spatial(64, i_0 * 4 + i_1)
                            vj = T.axis.spatial(64, j_0 * 4 + j_1)
                            vk = T.axis.reduce(64, k)
                            T.reads(A_global[vi, vk], B_global[vk, vj])
                            T.writes(C_global[vi, vj])
                            with T.init():
                                C_global[vi, vj] = T.float32(0.0)
                            C_global[vi, vj] = C_global[vi, vj] + A_global[vi, vk] * B_global[vk, vj]
        for ax0, ax1 in T.grid(64, 64):
            with T.sblock("C_global"):
                v0, v1 = T.axis.remap("SS", [ax0, ax1])
                T.reads(C_global[v0, v1])
                T.writes(C[v0, v1])
                C[v0, v1] = C_global[v0, v1]
